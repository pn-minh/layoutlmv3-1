% SLIDE 3: HIERARCHICAL POSITION ENCODING (HPE)
% ============================================================================
\begin{frame}{Hierarchical Position Encoding (HPE)}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=3pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_hpe.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{Structural Hierarchy:} Encodes Line, Column, and Block indices for every word token.
        \item \textbf{Multi-Column Disambiguation:} Prevents text across adjacent columns from merging horizontally.
        \item \textbf{Residual Linear Projection:} Combines the 3-tier coordinates and projects to hidden dimension.
        \item \textbf{Adaptive ReZero Gating:} Initialized at zero weight to preserve pre-trained backbone representations.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LayoutLMv3 Multimodal Backbone}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=3pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_backbone.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{Unified Sequence:} Processes text subwords and visual image patches jointly in one sequence.
        \item \textbf{Model Capacities:} Evaluated on Base (12 layers, 133M) and Large (24 layers, 368M) architectures.
        \item \textbf{2D Spatial Relative Bias:} Enriches attention scores with horizontal and vertical distance offsets.
        \item \textbf{Dual Output States:} Emits contextualized word vectors and visual patch vectors for downstream heads.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 5: SEGMENT CONTEXT MODULE (SCM)
% ============================================================================
\begin{frame}{Segment Context Module (SCM)}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=3pt] {
          \includegraphics[width=0.94\linewidth,height=0.65\textheight,keepaspectratio]{images/crop_scm.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{2.5pt}
        \item \textbf{Entity Fragmentation Fix:} Prevents multi-word entities from fracturing into inconsistent labels.
        \item \textbf{Segment Mean Pooling:} Aggregates individual word tokens into line-level segment representations.
        \item \textbf{Inter-Segment Transformer:} Captures semantic dependencies across lines throughout the entire page.
        \item \textbf{ReZero Residual Gate:} Dynamically blends segment-level consensus without losing word details.
        \item \textbf{Late-Fusion Head:} Injects segment context and first-token tags prior to final classification.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 6: INTRA-LINE BOUNDARY LOSS
% ============================================================================
\begin{frame}{Intra-Line Boundary Loss}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=3pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_boundary.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{Span Merging Prevention:} Separates adjacent key-value pairs that share the same horizontal line.
        \item \textbf{Pairwise Representation:} Evaluates adjacent token pairs along the reading line with an MLP.
        \item \textbf{Binary Objective:} Directly penalizes false merges between distinct semantic entities.
        \item \textbf{Zero Inference Latency:} Auxiliary training-only objective that adds no runtime overhead at test time.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 7: CROSS-MODAL CONTRASTIVE LOSS
% ============================================================================
\begin{frame}{Cross-Modal Contrastive Loss}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=3pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_contrastive.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \textbf{Spatial InfoNCE Alignment:} Pulls text tokens toward their overlapping visual patches in shared space.
        \item \textbf{Pre-SCM Decoupling:} Grounded on backbone tokens before segment pooling to prevent objective conflicts.
        \item \textbf{Detached Visual Anchors:} Freezes patch representations to prevent visual encoder drift during fine-tuning.
        \item \textbf{Calibrated Loss Weighting:} Scaled to low weights (0.03 for FUNSD, 0.01 for CORD) to regularize cleanly.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \textbf{Unified Loss Formulation:} Combines classification, boundary, and contrastive objectives together.
        \item \textbf{Complementary Roles:} Classification sets labels, boundary sharpens spans, contrastive grounds layout.
        \item \textbf{Joint Optimization:} End-to-end backpropagation allows all modules to mutually reinforce each other.
        \item \textbf{Best Checkpoint Loading:} Saves the peak validation F1 model to eliminate late-epoch noise.
      \end{itemize}
    \end{column}
