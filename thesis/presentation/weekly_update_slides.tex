      \scriptsize
      \begin{block}{\textbf{\color{PrimaryBlue}Structure-Aware LayoutLMv3}}
        Resolves the 4 primary structural bottlenecks of vanilla multimodal models:
      \end{block}
      \vspace{0.08cm}
      \begin{itemize}\setlength{\itemsep}{2.0pt}
        \item \textbf{1. Hierarchical Encoding (HPE):} Encodes Line, Column, and Block structure via adaptive $\beta$-gated residual projection.
        \item \textbf{2. Segment Context (SCM):} Eliminates entity fragmentation via segment pooling \& ReZero Transformer gating ($g$).
        \item \textbf{3. Boundary Loss ($\mathcal{L}_{\text{bound}}$):} Prevents adjacent key--value merging via pairwise line boundary BCE.
        \item \textbf{4. Contrastive Loss ($\mathcal{L}_{\text{cont}}$):} Grounds text tokens with visual image patches via spatial InfoNCE alignment.
        \item \textbf{Multi-Task Training ($\mathcal{L}_{\text{total}}$):} Unified loss with zero inference runtime penalty.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Component 1: Input Representation \& Embeddings}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_input.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Multimodal Triplet Inputs:}
          \begin{itemize}\scriptsize\setlength{\itemsep}{1.5pt}
            \item \textbf{Document Image:} Normalized $224 \times 224 \times 3$ RGB page.
            \item \textbf{OCR Text Words:} Subwords $w_i$ via RoBERTa BPE tokenizer ($V = 50{,}265$).
            \item \textbf{2D Bounding Boxes:} Integer coordinates $[x_0, y_0, x_1, y_1] \in [0, 1000]^4$.
          \end{itemize}
        \item \textbf{Visual Patch Embeddings:} Split into $14 \times 14 = 196$ patches ($16 \times 16$ px), linearly projected to $d$ plus learned 1D/2D visual positions.
        \item \textbf{Text \& Spatial Embeddings:}
          \begin{itemize}\scriptsize\setlength{\itemsep}{1.5pt}
            \item \textbf{1D Position:} Standard sequential token order ($0 \le i < 512$).
            \item \textbf{2D Spatial Position:} Four learned coordinate embedding tables for $(x_0, y_0, x_1, y_1)$ concatenated to represent box layout.
          \end{itemize}
        \item \textbf{Limitation of Standard Inputs:}
          Vanilla bounding boxes treat words as isolated geometric boxes, missing \textbf{reading lines, columns, or layout hierarchy}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 4: COMPONENT 2 - HIERARCHICAL POSITION ENCODING (HPE)
% ============================================================================
\begin{frame}{Component 2: Hierarchical Position Encoding (HPE)}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_hpe.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Motivation \& Vanilla Failure Case:}
          In multi-column tables, words with identical vertical $y$-coordinates in adjacent columns get improperly merged into one horizontal reading sequence.
        \item \textbf{Three-Tier Structural Triplet:}
          Extracts structural layout IDs for every subword token $i$:
          $$\mathbf{p}_{\text{hier}}(i) = (\text{Line ID } l_i, \text{ Column ID } c_i, \text{ Block ID } b_i)$$
        \item \textbf{Concatenation \& Projection ($W_{\text{hier}}$):}
          $$\mathbf{E}_{\text{hier}}(i) = W_{\text{hier}} \cdot [\mathbf{E}_{\text{line}}(l_i) \,\|\, \mathbf{E}_{\text{col}}(c_i) \,\|\, \mathbf{E}_{\text{block}}(b_i)]$$
          Projected linearly from $3 \times d_{\text{coord}}$ back to model dimension $d$.
        \item \textbf{Adaptive ReZero Gate ($\beta$):}
          $$\mathbf{E}_{\text{input}}(i) = \mathbf{E}_{\text{standard}}(i) + \beta \cdot \mathbf{E}_{\text{hier}}(i)$$
          Initialized at $\beta = 0.0$: ensures zero disruption to pretrained weights at step 0, gradually learning geometric hierarchy.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 5: COMPONENT 3 - MULTIMODAL TRANSFORMER BACKBONE
% ============================================================================
\begin{frame}{Component 3: LayoutLMv3 Multimodal Backbone}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_backbone.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.53\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Unified Multimodal Sequence:}
          Combines $L$ text subwords ($L \le 512$) and $196$ image patches into a single sequence of length $L + 196$:
          $$\mathbf{Z}^{(0)} = [\mathbf{t}_1, \dots, \mathbf{t}_L, \mathbf{v}_1, \dots, \mathbf{v}_{196}]$$
        \item \textbf{Transformer Architecture:}
          \begin{itemize}\scriptsize\setlength{\itemsep}{1.5pt}
            \item \texttt{base}: 12 layers, 12 heads, hidden size $d = 768$ (133M params).
            \item \texttt{large}: 24 layers, 16 heads, hidden size $d = 1024$ (368M params).
          \end{itemize}
        \item \textbf{2D Spatial Relative Attention Bias:}
          Enriches self-attention scores with relative horizontal and vertical distance offsets between token positions:
          $$\alpha_{ij} = \frac{\mathbf{q}_i \mathbf{k}_j^\top}{\sqrt{d_k}} + \mathbf{b}_x(x_i - x_j) + \mathbf{b}_y(y_i - y_j)$$
        \item \textbf{Output Representations:}
          Emits contextualized text vectors $\mathbf{h}_i \in \mathbb{R}^d$ and spatial visual patch vectors $\mathbf{v}_p \in \mathbb{R}^d$ for downstream reasoning.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 6: COMPONENT 4 - SEGMENT CONTEXT MODULE (SCM)
% ============================================================================
\begin{frame}{Component 4: Segment Context Module (SCM)}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_scm.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{The Entity Fragmentation Bottleneck:}
          Standard token classifiers evaluate each word in isolation, causing multi-word entities (e.g., long company names) to split into erratic BIO sequences.
        \item \textbf{Segment Mean Pooling:}
          Aggregates all subword tokens in segment $S_k = \{i_1, \dots, i_m\}$ into a single segment representation:
          $$\mathbf{s}_k = \frac{1}{|S_k|} \sum_{i \in S_k} \mathbf{h}_i$$
        \item \textbf{Inter-Segment Transformer Context ($\mathbf{c}_k$):}
          Encodes the sequence of segments $[\mathbf{s}_1, \dots, \mathbf{s}_K]$ via multi-head self-attention to capture cross-line document relationships.
        \item \textbf{ReZero Residual Segment Gate ($g$):}
          $$\mathbf{s}'_k = \mathbf{s}_k + g \cdot (\mathbf{c}_k - \mathbf{s}_k), \quad g \leftarrow 0.0$$
        \item \textbf{Late-Fusion \& Classifier:}
          Injects segment consensus $\mathbf{s}'_k$ and a learned first-token indicator ($B\text{-} / I\text{-}$) back into word representations before Softmax prediction.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 7: COMPONENT 5 - INTRA-LINE BOUNDARY LOSS
% ============================================================================
\begin{frame}{Component 5: Intra-Line Boundary Loss ($\mathcal{L}_{\text{bound}}$)}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_boundary.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{The Adjacent Span Merging Problem:}
          When a key (Question) and value (Answer) appear horizontally on the same line (e.g., \texttt{"INVOICE NO: 10482"}), models often falsely predict a single continuous entity span.
        \item \textbf{Pairwise Boundary Formulation:}
          Evaluates consecutive tokens $(i, i+1)$ that share the same horizontal reading line:
          $$\hat{y}_i^{\text{bound}} = \sigma\left(\mathbf{W}_b \cdot \text{GELU}(\mathbf{W}_p [\mathbf{h}_i \,\|\, \mathbf{h}_{i+1}])\right)$$
        \item \textbf{Binary Supervised Objective:}
          $y_i = 1$ if an entity boundary occurs between token $i$ and $i+1$, and $y_i = 0$ if they belong to the same entity:
          $$\mathcal{L}_{\text{bound}} = -\frac{1}{|\mathcal{B}|} \sum_{i \in \mathcal{B}} [y_i \log \hat{y}_i + (1 - y_i) \log (1 - \hat{y}_i)]$$
        \item \textbf{Zero Inference Runtime Overhead:}
          Operates strictly as an auxiliary regularization loss during training ($\lambda_{\text{bound}} = 0.1$); stripped away at inference time.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 8: COMPONENT 6 - LATEST CONTRASTIVE LOSS IMPLEMENTATION
% ============================================================================
\begin{frame}{Component 6: Latest Contrastive Loss ($\mathcal{L}_{\text{cont}}$)}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_contrastive.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.2pt}
        \item \textbf{Spatial-Overlap InfoNCE Objective:}
          Pulls text token representations toward their overlapping visual image patches while pushing non-overlapping background patches away:
          $$\mathcal{L}_{\text{cont}} = -\sum_{i} \sum_{p \in \mathcal{P}^+(i)} w_{ip} \log \frac{\exp(\mathbf{z}_i^{\text{text}} \cdot \mathbf{z}_p^{\text{img}} / \tau)}{\sum_{k} \exp(\mathbf{z}_i^{\text{text}} \cdot \mathbf{z}_k^{\text{img}} / \tau)}$$
        \item \textbf{Fine-Tuning Innovation 1: Pre-SCM Decoupling:}
          Computed on pre-segment backbone tokens $\mathbf{h}_i$ rather than post-SCM vectors. \textbf{Completely eliminates the objective conflict} between SCM entity pooling and spatial patch pulling!
        \item \textbf{Fine-Tuning Innovation 2: Detached Visual Anchors:}
          Visual patch features are detached ($\mathbf{v}_p.\text{detach}()$), fixing them as invariant geometric anchors and preventing vision encoder drift.
        \item \textbf{Fine-Tuning Innovation 3: Balanced Weighting:}
          Tuned from excessive $0.1$ down to $\lambda_{\text{cont}} = \mathbf{0.03}$ (FUNSD) and $\mathbf{0.01}$ (CORD) with $\tau = \mathbf{0.10}$ to prevent dominating classification loss.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 9: COMPONENT 7 - MULTI-TASK TRAINING OBJECTIVE
% ============================================================================
\begin{frame}{Component 7: Multi-Task Training Objective}
  \vspace{-0.10cm}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.94\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_loss_objective.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Unified Multi-Task Loss Formulation:}
          $$\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{cls}} + \lambda_{\text{bound}} \mathcal{L}_{\text{bound}} + \lambda_{\text{cont}} \mathcal{L}_{\text{cont}}$$
        \item \textbf{Complementary Roles of Loss Terms:}
          \begin{itemize}\scriptsize\setlength{\itemsep}{1.5pt}
            \item \textbf{$\mathcal{L}_{\text{cls}}$ (Cross-Entropy):} Primary objective; establishes global semantic classification decision boundaries.
            \item \textbf{$\mathcal{L}_{\text{bound}}$ (BCE):} Enforces sharp local transition boundaries between adjacent question/answer entities.
            \item \textbf{$\mathcal{L}_{\text{cont}}$ (Spatial InfoNCE):} Regularizes cross-modal alignment between subwords and visual layout patches.
          \end{itemize}
        \item \textbf{Synergistic Gradient Flow:}
          Backpropagating all three objectives jointly guarantees that visual grounding and boundary awareness directly benefit final token classification.
        \item \textbf{Optimal Checkpoint Loading:}
          Trained with \texttt{load\_best\_model\_at\_end} on validation F1 score, eliminating late-epoch training drift.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
