% SLIDE 3: HIERARCHICAL POSITION ENCODING (HPE)
% ============================================================================
\begin{frame}{Hierarchical Position Encoding (HPE)}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.96\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_hpe.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Internal Model Mechanics:} Embeds discrete 3-tier coordinate tuples $(i_{\text{line}}, i_{\text{col}}, i_{\text{blk}})$ via linear layers and adds them to token embeddings via a learnable ReZero scalar gate: $\mathbf{x}_i = \mathbf{x}_i^{\text{base}} + \beta \mathbf{e}_i^{\text{HPE}}$.
        \item \textbf{Example 1 (Multi-Column Invoices):} Without HPE, horizontal self-attention confuses text across neighboring table columns at the same vertical height; with HPE, distinct column indices enforce strict vertical separation.
        \item \textbf{Example 2 (Multi-Line Wrapped Entities):} Without HPE, long multi-line headers fracture into disconnected tokens; with HPE, shared block indexing anchors wrapped lines to the same entity parent.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================================================================
% SLIDE 4: LAYOUTLMV3 MULTIMODAL BACKBONE
% ============================================================================
\begin{frame}{LayoutLMv3 Multimodal Backbone}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.96\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_backbone.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Internal Model Mechanics:} Jointly contextualizes subword text tokens and visual patches ($16 \times 16$) in a shared self-attention sequence, enriched by 2D relative spatial position bias $\mathbf{B}_{2D}(\Delta x, \Delta y)$ across transformer layers.
        \item \textbf{Example 1 (Degraded / Faded Receipts):} Without visual fusion, corrupted or missing OCR characters fail text-only classification; with multimodal fusion, high-level image patches recover the true field semantics.
        \item \textbf{Example 2 (Repeated Labels in Forms):} Without 2D spatial bias, identical words (e.g. "Date" in header vs. signature line) generate identical representations; with 2D bias, distinct coordinates disambiguate their roles.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 5: SEGMENT CONTEXT MODULE (SCM)
% ============================================================================
\begin{frame}{Segment Context Module (SCM)}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.96\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_scm.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Internal Model Mechanics:} Pools token states into line-level segment vectors, propagates inter-line context via bidirectional recurrence, and injects segment consensus into token logits via gated late-fusion: $\mathbf{L}_i = \mathbf{L}_i^{\text{seg}} + M_i \cdot \alpha (\mathbf{L}_i^{\text{tok}} - \mathbf{L}_i^{\text{seg}})$.
        \item \textbf{Example 1 (Distant Key-Value Association):} Without SCM, attention between "Invoice Total" at top-right and the numerical total at bottom-right dilutes across 512 tokens; with SCM, inter-segment recurrence preserves global pairing.
        \item \textbf{Example 2 (Dense Itemized Tables):} Without SCM, token predictions suffer from local label jitter across rows; with SCM, segment consensus enforces uniform semantic classification across the entire line segment.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 6: INTRA-LINE BOUNDARY LOSS
% ============================================================================
\begin{frame}{Intra-Line Boundary Loss}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.96\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_boundary.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Internal Model Mechanics:} Feeds adjacent token hidden pairs $(\mathbf{h}_i, \mathbf{h}_{i+1})$ along the horizontal reading sequence into an auxiliary MLP to supervise entity transition boundaries via binary cross-entropy, adding zero test-time latency.
        \item \textbf{Example 1 (Inline Key-Value Fields):} Without Boundary Loss, "Tel: 555-0199" suffers from label bleed where the number's prefix is misclassified as key; with Boundary Loss, transition penalties enforce an exact split.
        \item \textbf{Example 2 (Dense Multi-Field Lines):} Without Boundary Loss, adjacent receipt entries \texttt{[Item] [Qty] [Price]} merge into a single entity span; with Boundary Loss, boundary supervision maintains clean field separation.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 7: CROSS-MODAL CONTRASTIVE LOSS
% ============================================================================
\begin{frame}{Cross-Modal Contrastive Loss}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}
        \node[draw=BorderGray, fill=white, rounded corners=6pt, line width=0.8pt, inner sep=2pt] {
          \includegraphics[width=0.96\linewidth,height=0.68\textheight,keepaspectratio]{images/crop_contrastive.png}
        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Internal Model Mechanics:} Projects text tokens prior to SCM pooling and aligns them with detached visual patch anchors via InfoNCE loss $\mathcal{L}_{\text{cont}} = -\log \frac{\exp(\mathbf{z}_i^t \cdot \mathbf{z}_i^{v+} / \tau)}{\sum_j \exp(\mathbf{z}_i^t \cdot \mathbf{z}_j^v / \tau)}$, preventing gradient conflict.
        \item \textbf{Example 1 (Logos / Decorative Headers):} Without Contrastive Loss, strong visual patch features dominate text embeddings and cause representation drift; with Contrastive Loss, normalized metric projection stabilizes alignment.
        \item \textbf{Example 2 (Stamps and Watermarks):} Without Contrastive Loss, background clutter creates spurious token-patch attention correlations; with Contrastive Loss, negative patch repulsion teaches the model to ignore background noise.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

        };
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3.5pt}
        \item \textbf{Internal Model Mechanics:} Jointly minimizes $\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{CE}} + \lambda_{\text{bnd}}\mathcal{L}_{\text{bnd}} + \lambda_{\text{cont}}\mathcal{L}_{\text{cont}}$ ($\lambda_{\text{bnd}}=0.10, \lambda_{\text{cont}}=0.01 \dots 0.03$), updating all transformer parameters simultaneously through unified backpropagation.
        \item \textbf{Example 1 (Isolated Codes / Abbreviations):} Without Joint Objective, standard cross-entropy overfits to lexical frequency and misclassifies rare codes; with Joint Objective, boundary and contrastive losses act as structural regularizers.
        \item \textbf{Example 2 (Unseen Layout Shifts):} Without Joint Objective, extraction F1 degrades sharply on unseen document templates; with Joint Objective, multi-task gradients preserve structural invariants across layout variations.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{In-Depth Ablation Insights \& Analysis}
  \vspace{-0.10cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \textbf{Layout Complexity Sensitivity (HPE):} FUNSD drops more ($-1.00\%$) than CORD ($-0.75\%$) without HPE because complex forms feature nested blocks, whereas receipts follow simpler single-column flows.
        \item \textbf{Dominant Contribution of SCM:} SCM produces the largest ablation drop on FUNSD ($-2.34\%$ Large, $-1.79\%$ Base), confirming that long-distance key-value propagation is the primary bottleneck in form extraction.
        \item \textbf{Boundary Precision vs. Speed:} Intra-line boundary supervision eliminates span bleeding between adjacent fields with zero runtime overhead, recovering $+0.95\% \dots +1.20\%$ F1.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \textbf{Contrastive Gradient Resolution:} Decoupling text projections before SCM and detaching visual anchors transformed contrastive loss from a $-0.32\%$ drag into a $+0.90\%$ gain across all models.
        \item \textbf{Architectural Synergy:} The total F1 gain of the full system ($+3.5\% \dots +4.2\%$ over baseline) exceeds individual isolated module sums, demonstrating strong cross-module synergy.
        \item \textbf{Scale Invariance:} Relative module importance remains consistent across both Base (12-layer) and Large (24-layer) models, confirming architectural generalization.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
