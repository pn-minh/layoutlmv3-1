% Macro to draw a 3D isometric block (exact HEADS Fig. 1 style)
\newcommand{\drawblock}[5]{% {x}{y}{w}{h}{color}
    \pgfmathsetmacro{\x}{#1}
    \pgfmathsetmacro{\y}{#2}
    \pgfmathsetmacro{\w}{#3}
    \pgfmathsetmacro{\h}{#4}
    \fill[#5!80!black] (\x+\w, \y) -- (\x+\w+0.12, \y+0.12) -- (\x+\w+0.12, \y+\h+0.12) -- (\x+\w, \y+\h) -- cycle;
    \fill[#5!55!white] (\x, \y+\h) -- (\x+0.12, \y+\h+0.12) -- (\x+\w+0.12, \y+\h+0.12) -- (\x+\w, \y+\h) -- cycle;
    \fill[#5] (\x, \y) rectangle (\x+\w, \y+\h);
    \draw[black!65, very thin] (\x, \y) rectangle (\x+\w, \y+\h);
    \draw[black!65, very thin] (\x+\w, \y) -- (\x+\w+0.12, \y+0.12) -- (\x+\w+0.12, \y+\h+0.12) -- (\x+\w, \y+\h);
    \draw[black!65, very thin] (\x, \y+\h) -- (\x+0.12, \y+\h+0.12) -- (\x+\w+0.12, \y+\h+0.12);
}

% Macro to draw a contiguous 3-block 3D bar (exact HEADS Fig. 1 embedding bar)
\newcommand{\drawbar}[5]{% {x}{y}{seg_w}{h}{color}
    \foreach \i in {0,1,2} {
        \pgfmathsetmacro{\cx}{#1 + \i*(#3)}
        \drawblock{\cx}{#2}{#3}{#4}{#5}
    }
}

% Macro for clipped circular Adder (+) where inner cross touches outer circle exactly
\newcommand{\drawadder}[4]{% {name}{x}{y}{color}
    \node[circle, draw=#4, fill=white, line width=0.60mm, minimum size=0.72cm,
        path picture={
            \draw[line width=0.55mm, #4]
                (path picture bounding box.north) -- (path picture bounding box.south)
                (path picture bounding box.west) -- (path picture bounding box.east);
        }] (#1) at (#2, #3) {};
}
